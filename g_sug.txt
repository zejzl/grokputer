import asyncio
from .reasoner import Reasoner

class Validator(Reasoner):
    async def critique_output(self, output: str, threshold: float = 0.95) -> bool:
        prompt = f"Validate this output for consistency/accuracy (score 0-1): {output}"
        score = float(await self.query_llm(prompt))  # Parse score
        if score < threshold:
            await self.refine_output(output, score)  # Human-in-loop or auto
        return score >= threshold
