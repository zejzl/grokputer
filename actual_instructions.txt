textimport pyautogui
from PIL import Image
import base64
from io import BytesIO

def get_screenshot():
    img = pyautogui.screenshot()
    buffered = BytesIO()
    img.save(buffered, format="PNG")
    return base64.b64encode(buffered.getvalue()).decode('utf-8')
