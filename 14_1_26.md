**Example Complete Workflow:**
```python
from src.workflow import Workflow
from src.workflow.nodes import HTTPNode, AINode, ConditionalNode, SlackNode, AsanaNode

# Create automated task management workflow
workflow = Workflow("task_processor")

# 1. Fetch data from external API
http_node = HTTPNode("fetch_data", config={
    "url": "https://api.example.com/tasks",
    "method": "GET"
})

# 2. AI analysis and decision
ai_node = AINode("analyze_task", config={
    "provider": "grok",
    "prompt": "Analyze this task: {{response_body}}. Is it urgent?",
    "output_format": "decision"
})

# 3. Conditional branching based on urgency
conditional_node = ConditionalNode("check_urgency", config={
    "conditions": [{"field": "ai_response", "operator": "==", "value": True}],
    "logic": "and"
})

# 4. Create Asana task for urgent items
asana_node = AsanaNode("create_task", config={
    "operation": "create_task",
    "workspace_gid": "12345",
    "name": "Urgent: {{task_name}}",
    "due_date": "2026-01-20"
})

# 5. Send Slack notification
slack_node = SlackNode("notify_team", config={
    "operation": "send_message",
    "channel": "#alerts",
    "text": "🚨 Urgent task created: {{task_name}}"
})

# Connect workflow
workflow.add_node(http_node)
workflow.add_node(ai_node)
workflow.add_node(conditional_node)
workflow.add_node(asana_node)
workflow.add_node(slack_node)

workflow.add_edge(http_node, ai_node)
workflow.add_edge(ai_node, conditional_node)
workflow.add_edge(conditional_node, asana_node, condition="true")
workflow.add_edge(asana_node, slack_node)

# Execute
result = await workflow.run({"task_name": "Fix critical bug"})
```
